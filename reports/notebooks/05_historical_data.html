A reproducible held-out slice¶

The final 14 days form the scoring period. Fit the model using only the preceding 100 days. Held-out intervention schedules are scenario inputs; future case counts are reserved for evaluation. Negative daily revisions are clipped and the first training difference is zero.

In [3]:
sample = series.tail(114).copy()
train, test = sample.iloc[:-14], sample.iloc[-14:]
columns = list(em.NPI_COLUMNS)
inputs = sample[columns].ffill().fillna(0).to_numpy(float).T
# Read the same bundled population table used by MATLAB.
populations = pd.read_csv(root / "data/populations.csv").fillna({"RegionName": ""})
population = float(
    populations[
        (populations.CountryName == country) & (populations.RegionName == "")
    ].Population2020.iloc[0]
)
model = em.fit_npi_model(
    train.ConfirmedCases.to_numpy(), inputs[:, :-14], population, em.NPI_MAXES
)
predicted, _ = em.forecast_npi(model, inputs[:, -14:])
actual = np.maximum(
    0, np.diff(np.r_[train.ConfirmedCases.iloc[-1], test.ConfirmedCases.to_numpy()])
)
error = predicted - actual
display(
    pd.DataFrame(
        {
            "Days": [len(test)],
            "MAE": [np.mean(np.abs(error))],
            "RMSE": [np.sqrt(np.mean(error**2))],
            "Bias": [np.mean(error)],
        }
    ).round(2)
)
fig, ax = plt.subplots()
ax.plot(test.Date, actual, label="Held-out daily reports")
ax.plot(test.Date, predicted, label="Known-policy forecast")
ax.set(title="France · a held-out forecast", xlabel="Date", ylabel="Cases / day")
ax.legend()
plt.show()
